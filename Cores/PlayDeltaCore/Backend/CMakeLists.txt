# Record actual configured archive targets, avoiding stale outputs after upgrades.
function(collect_archives directory result)
    get_property(targets DIRECTORY "${directory}" PROPERTY BUILDSYSTEM_TARGETS)
    set(archives "")
    foreach(target IN LISTS targets)
        get_target_property(kind "${target}" TYPE)
        if(kind STREQUAL "STATIC_LIBRARY")
            list(APPEND archives "$<TARGET_FILE:${target}>")
        endif()
    endforeach()
    get_property(children DIRECTORY "${directory}" PROPERTY SUBDIRECTORIES)
    foreach(child IN LISTS children)
        collect_archives("${child}" child_archives)
        list(APPEND archives ${child_archives})
    endforeach()
    set(${result} "${archives}" PARENT_SCOPE)
endfunction()
collect_archives("${CMAKE_CURRENT_SOURCE_DIR}" archives)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/$<CONFIG>-archives.txt" CONTENT "$<JOIN:${archives},\n>\n")
